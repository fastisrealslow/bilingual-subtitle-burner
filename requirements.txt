# bilingual-subtitle-burner 依赖
#
# 安装：
#   pip install -r requirements.txt
# 注意：faster-whisper 1.0.3 会锁定 av<13，缺少 ffmpeg 开发库时编译失败，
#       因此固定 >=1.1.0，并建议用 --only-binary 安装：
#   pip install --only-binary=:all: "faster-whisper>=1.1.0"
#
# 系统依赖（需自行安装，不在 pip 内）：
#   ffmpeg  —— 切片、转码、截帧（brew install ffmpeg / apt install ffmpeg）
#   yt-dlp  —— 视频采集（可 pip 安装，见下）

faster-whisper>=1.1.0
# 取 YouTube 源时裸装这一行不够：yt-dlp 自 2025.11.12 起要靠 EJS（外部 JS 运行时
# + 求解脚本）解 n 签名挑战，缺了只能看到极少数格式，取源退成「Requested format
# is not available」。补装：
#   pip install --upgrade "yt-dlp[default,deno]"
# [default] 带 yt-dlp-ejs 求解脚本，[deno] 带运行时本体（约 100MB，只跑测试的话
# 用不着）。produce.yml 的「安装 yt-dlp 与 EJS 依赖」那步装的就是这个。
yt-dlp>=2024.0.0
Pillow>=10.0.0
requests>=2.31.0

# 繁转简：Whisper 对中文源常输出繁体（实测台湾财经视频出「公佈」「調到了」），
# 而 B站/抖音受众以简体为主，统一转一道。
zhconv>=1.4.3
# 封面标题按词折行，避免把「反常识」从中间劈开。缺失时会退化成逐字折行。
jieba>=0.42.1
# 封面选帧前的人脸预筛（step7）：haar 级联本地过一遍，只把有正脸的帧送 vision。
# headless 版不拉 libGL 等 GUI 依赖，CI runner 上装得动。缺失时自动跳过预筛。
# 必须锁 <5：OpenCV 5.0 移除了 cv2.CascadeClassifier，且 cv2/data 里不再随包
# 附带 haarcascade_*.xml，装上 5.x 会让预筛静默退化成「全部帧都送 vision」。
opencv-python-headless>=4.8.0,<5

# 封面标题字体：NotoSansCJK-Bold.ttc 是一个 TTC 合集，PIL 不指定 index 时默认
# 取第 0 个子字体（Noto Sans CJK JP），不是 SC。用 fontTools 按 name 表在合集里
# 查出「Noto Sans CJK SC」的真实下标，避免简体标题里混入日文字形变体。
# 缺失时退化成 index=0（旧行为，不会比修之前更差，但拿不到 SC 子字体）。
fonttools>=4.0.0
